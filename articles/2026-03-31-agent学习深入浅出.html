现在大家都在频繁的使用各种各样的ai，例如deepseek 元宝 豆包什么的，这些都是以聊天的形式使用。

上下文工程

有了如上的基础，我们会发现一个现象，当agent阅读一个文件后，会在记忆(对话记录)中产生大量的上下文占用，当执行了几百轮的对话后，第一次打开的文件实际上已经不是很重要了，但是他还是占据着我们的上下文，这就会导致token数量激增，即以注意力被分散。

于是乎我们需要对一些用处不大的记忆进行压缩，达到节约上下文成本的效果。不过现在有缓存，前面的对话记录不变只是在末尾新增对话会命中缓存，价格会便宜很多，有时候还不需要压缩了。但是这种上下文工程的思路需要理解。

 

1.对agent执行任务真正有用的可能只有最近10次甚至几次的工具调用结果，更加早的调用结果实际上可以被压缩，模型只需要知道曾经调用过，但是没什么用即可(但是这个清理的度很难把握)。

2.很早很早的工具调用结果实际上是可以被压缩的，例如100轮之前的东西，就像人类大脑一样只记得大概，agent也是如此，给它知道自己曾经尝试过哪些工具即可。

3.给与agent一个工具，这个工具用于压缩当前的上下文，让它自己决定自己的思路是否需要被压缩。

#!/usr/bin/env python3
"""
ReAct-based CLI Agent using AsyncAnthropic SDK
"""

import asyncio
import json
import os
import subprocess
from typing import Any

from anthropic import AsyncAnthropic
from dotenv import load_dotenv

# Load environment variables from .env file
load_dotenv()

# Configuration
ANTHROPIC_API_KEY = os.getenv("ANTHROPIC_API_KEY")
ANTHROPIC_BASE_URL = os.getenv("ANTHROPIC_BASE_URL")
MODEL_ID = os.getenv("MODEL_ID", "claude-sonnet-4-6")

# System Prompt - require agent to use tools, no explanations
SYSTEM_PROMPT = """You are a ReAct agent. Use the provided tools to solve tasks.
Execute commands and observe results to complete objectives.
Do not explain your reasoning. Just use tools and provide the final answer."""


# Tool Definition - Unconstrained Bash Tool
BASH_TOOL = {
    "name": "bash",
    "description": "Execute any bash command on the system. Use this tool to run shell commands.",
    "input_schema": {
        "type": "object",
        "properties": {
            "command": {
                "type": "string",
                "description": "The bash command to execute"
            }
        },
        "required": ["command"]
    }
}


class CLIAgent:
    """ReAct-based CLI Agent with async control loop."""

    def __init__(self):
        self.client = AsyncAnthropic(
            api_key=ANTHROPIC_API_KEY,
            base_url=ANTHROPIC_BASE_URL
        )
        self.messages: list[dict[str, Any]] = []

    async def execute_bash(self, command: str) -> str:
        """Execute a bash command and return the output."""
        try:
            result = subprocess.run(
                command,
                shell=True,
                capture_output=True,
                text=True,
                timeout=300
            )
            output = result.stdout
            if result.stderr:
                output += f"\nSTDERR: {result.stderr}"
            if result.returncode != 0:
                output += f"\nExit Code: {result.returncode}"
            return output.strip() if output.strip() else "(no output)"
        except subprocess.TimeoutExpired:
            return "Error: Command timed out after 300 seconds"
        except Exception as e:
            return f"Error: {str(e)}"

    async def process_tool_use(self, tool_name: str, tool_input: dict) -> str:
        """Process a tool use request and return the result."""
        if tool_name == "bash":
            command = tool_input.get("command", "")
            print(f"\n🔧 Executing: {command}")
            result = await self.execute_bash(command)
            print(f"📤 Result:\n{result}\n")
            return result
        else:
            return f"Error: Unknown tool '{tool_name}'"

    async def run_loop(self, user_input: str) -> str:
        """
        Main async control loop following ReAct pattern.
        - Request LLM with user input
        - Check stop_reason for tool_use
        - Execute tools and append results to messages
        - Continue until no more tool calls
        """
        # Initialize messages with user input
        self.messages = [
            {"role": "user", "content": user_input}
        ]

        while True:
            # Request LLM
            response = await self.client.messages.create(
                model=MODEL_ID,
                max_tokens=4096,
                system=SYSTEM_PROMPT,
                tools=[BASH_TOOL],
                messages=self.messages
            )

            # Check stop_reason
            stop_reason = response.stop_reason

            # Append assistant response to messages
            self.messages.append({
                "role": "assistant",
                "content": response.content
            })

            # If tool_use, process and continue loop
            if stop_reason == "tool_use":
                # Find and process all tool_use blocks
                tool_results = []

                for block in response.content:
                    if block.type == "tool_use":
                        tool_name = block.name
                        tool_input = block.input
                        tool_id = block.id

                        # Execute the tool
                        result = await self.process_tool_use(tool_name, tool_input)

                        # Build tool result following Anthropic tool-use spec
                        tool_results.append({
                            "type": "tool_result",
                            "tool_use_id": tool_id,
                            "content": result
                        })

                # Append tool results as user message
                self.messages.append({
                    "role": "user",
                    "content": tool_results
                })

                # Continue the loop for next iteration
                continue

            # No more tool calls - extract and return final answer
            final_answer = ""
            for block in response.content:
                if hasattr(block, 'text'):
                    final_answer += block.text

            return final_answer.strip() if final_answer.strip() else "Task completed."

    async def run_interactive(self):
        """Run the interactive CLI session."""
        print("=" * 50)
        print("ReAct CLI Agent (AsyncAnthropic)")
        print(f"Model: {MODEL_ID}")
        print("Type 'exit' or 'quit' to stop")
        print("=" * 50)

        while True:
            try:
                # Get user input
                user_input = input("\n👤 You: ").strip()

                if not user_input:
                    continue

                if user_input.lower() in ("exit", "quit"):
                    print("\n👋 Goodbye!")
                    break

                # Run the agent loop
                print("\n🤖 Agent thinking...")
                result = await self.run_loop(user_input)

                # Print final answer
                print(f"\n✅ Final Answer:\n{result}")

            except KeyboardInterrupt:
                print("\n\n👋 Goodbye!")

def parse_skill_metadata(skill_file: Path) -> dict | None:
    """
    Parse a skill markdown file and extract metadata from frontmatter.

    Expected format:
    ---
    name: skill_name
    description: Short description of the skill
    ---

    # Skill Content
    ...

    Returns:
        dict with 'description' and 'file_path', or None if invalid
    """
    try:
        content = skill_file.read_text(encoding='utf-8')

        # Check for frontmatter
        if not content.startswith("---"):
            # No frontmatter, use filename as name and first line as description
            lines = content.strip().split('\n')
            description = lines[0].lstrip('# ').strip()[:100] if lines else skill_file.stem
            return {
                "description": description or f"Skill: {skill_file.stem}",
                "file_path": str(skill_file)
            }

        # Parse frontmatter
        end_idx = content.find("---", 3)
        if end_idx == -1:
            return None

        frontmatter_text = content[3:end_idx].strip()
        metadata = {}

        for line in frontmatter_text.split('\n'):
            if ':' in line:
                key, value = line.split(':', 1)
                metadata[key.strip()] = value.strip()

        # Extract description from frontmatter or content
        description = metadata.get('description', '')
        if not description:
            # Try to get first heading or paragraph
            body = content[end_idx + 3:].strip()
            lines = [l for l in body.split('\n') if l.strip() and not l.strip().startswith('#')]
            if lines:
                description = lines[0].strip()[:100]
            else:
                description = f"Skill: {skill_file.stem}"

        return {
            "description": description,
            "file_path": str(skill_file),
            "name": metadata.get('name', skill_file.stem)
        }
    except Exception as e:
        print(f"Warning: Failed to parse skill file {skill_file}: {e}")
        return None

        except FileNotFoundError:
            return f"Error: File not found: {file_path}"
        except PermissionError:
            return f"Error: Permission denied: {file_path}"
        except Exception as e:
            return f"Error: {str(e)}"

        """Execute a bash command and return the output."""
        try:
            result = subprocess.run(
                command,
                shell=True,
                capture_output=True,
                text=True,
                timeout=300
            )
            output = result.stdout
            if result.stderr:
                output += f"\nSTDERR: {result.stderr}"
            if result.returncode != 0:
                output += f"\nExit Code: {result.returncode}"
            return output.strip() if output.strip() else "(no output)"
        except subprocess.TimeoutExpired:
            return "Error: Command timed out after 300 seconds"
        except Exception as e:
            return f"Error: {str(e)}"

    async def update_todo(self, tasks: list[dict]) -> str:
        """Update the global todo list and return status message."""

                TODO_LIST = []
                return "✅ Task list cleared."

        # Validate and process tasks
        validated_tasks = []
        errors = []

        for i, task in enumerate(tasks):
            if not isinstance(task, dict):
                errors.append(f"Task {i}: Not a valid object")
                continue

            description = task.get("description")
            status = task.get("status")

            if not description:
                errors.append(f"Task {i}: Missing 'description' field")
                continue

            if status not in VALID_TASK_STATES:
                errors.append(f"Task {i}: Invalid status '{status}'. Must be one of {VALID_TASK_STATES}")
                continue

            validated_tasks.append({
                "description": description,
                "status": status
            })

        if errors:
            error_msg = "; ".join(errors)
            return f"❌ Validation errors: {error_msg}"

        # Update global todo list
        TODO_LIST = validated_tasks

        # Build summary
        status_counts = {"等待": 0, "失败": 0, "完成": 0}
        for task in validated_tasks:
            status_counts[task["status"]] += 1

        summary = (f"✅ Todo list updated ({len(validated_tasks)} tasks)\n"

        # Print current task list
        print(f"\n📋 Current Todo List:")
        for i, task in enumerate(validated_tasks, 1):
            status_icon = {"等待": "⏳", "失败": "❌", "完成": "✅"}[task["status"]]

    async def run_loop(self, user_input: str) -> str:
        """
        Main async control loop following ReAct pattern.
        - Request LLM with user input
        - Check stop_reason for tool_use
        - Execute tools and append results to messages
        - Continue until no more tool calls
        - Monitor: Inject reminder if 3 consecutive rounds without update_todo
        """
        # Initialize messages with user input
        self.messages = [
            {"role": "user", "content": user_input}
        ]

        # Reset the consecutive no-update counter at the start of each session
        self.consecutive_no_todo_updates = 0

        # Get dynamic system prompt with current skills
        system_prompt = get_system_prompt()

        while True:
            # Request LLM
            response = await self.client.messages.create(
                model=MODEL_ID,
                max_tokens=4096,
                system=system_prompt,
                tools=TOOLS,
                messages=self.messages
            )

            # Check stop_reason
            stop_reason = response.stop_reason

            # Append assistant response to messages
            self.messages.append({
                "role": "assistant",
                "content": response.content
            })

            # If tool_use, process and continue loop
            if stop_reason == "tool_use":
                # Find and process all tool_use blocks
                tool_results = []

                # Track if update_todo was called in this round
                todo_updated_this_round = False

                for block in response.content:
                    if block.type == "tool_use":
                        tool_name = block.name
                        tool_input = block.input
                        tool_id = block.id

                        # Check if update_todo is being called
                        if tool_name == "update_todo":
                            todo_updated_this_round = True

                        # Execute the tool
                        result = await self.process_tool_use(tool_name, tool_input)

                        # Build tool result following Anthropic tool-use spec
                        tool_results.append({
                            "type": "tool_result",
                            "tool_use_id": tool_id,
                            "content": result
                        })

                # Update counter based on whether update_todo was called
                if todo_updated_this_round:
                    self.consecutive_no_todo_updates = 0
                else:
                    self.consecutive_no_todo_updates += 1

                # Check if we need to inject a reminder
                if self.consecutive_no_todo_updates >= self.TODO_REMINDER_THRESHOLD:
                    print(f"\n⚠️ WARNING: {self.consecutive_no_todo_updates} consecutive rounds without update_todo call")
                    print("📋 Injecting reminder to sync task progress...")

                    # Append tool results first
                    self.messages.append({
                        "role": "user",
                        "content": tool_results
                    })

                    # Inject reminder as a separate user message (not as tool_result)
                    reminder_msg = (
                        "【系统提醒】您已连续多轮推理未更新任务清单。\n"
                        "为了保持推理一致性，请立即使用 update_todo 工具同步当前任务进度。\n"
                        "当前状态：\n"
                        f"- 待处理任务数: {len([t for t in TODO_LIST if t['status'] == '等待'])}\n"
                        f"- 已完成任务数: {len([t for t in TODO_LIST if t['status'] == '完成'])}\n"
                        f"- 失败任务数: {len([t for t in TODO_LIST if t['status'] == '失败'])}\n"
                        "请审视当前进度并更新任务状态。"
                    )
                    self.messages.append({
                        "role": "user",
                        "content": reminder_msg
                    })

                    # Reset counter after reminder
                    self.consecutive_no_todo_updates = 0

                    # Continue the loop for next iteration
                    continue

                # Append tool results as user message
                self.messages.append({
                    "role": "user",
                    "content": tool_results
                })

                # Continue the loop for next iteration
                continue

            # No more tool calls - extract and return final answer
            final_answer = ""
            for block in response.content:
                if hasattr(block, 'text'):
                    final_answer += block.text

            return final_answer.strip() if final_answer.strip() else "Task completed."

def scan_skills_directory() -> dict[str, dict]:
    """

    skills = {}

        SKILLS_DIR.mkdir(parents=True, exist_ok=True)
        return skills

    for skill_file in SKILLS_DIR.glob("*.md"):
        skill_name = skill_file.stem  # filename without extension
        skill_info = parse_skill_metadata(skill_file)
        if skill_info:
            skills[skill_name] = skill_info

def parse_skill_metadata(skill_file: Path) -> dict | None:
    """
    Parse a skill markdown file and extract metadata from frontmatter.

    Expected format:
    ---
    name: skill_name
    description: Short description of the skill
    ---

    # Skill Content
    ...

    Returns:
        dict with 'description' and 'file_path', or None if invalid
    """
    try:
        content = skill_file.read_text(encoding='utf-8')

        # Check for frontmatter
        if not content.startswith("---"):
            # No frontmatter, use filename as name and first line as description
            lines = content.strip().split('\n')
            description = lines[0].lstrip('# ').strip()[:100] if lines else skill_file.stem
            return {
                "description": description or f"Skill: {skill_file.stem}",
                "file_path": str(skill_file)
            }

        # Parse frontmatter
        end_idx = content.find("---", 3)
        if end_idx == -1:
            return None

        frontmatter_text = content[3:end_idx].strip()
        metadata = {}

        for line in frontmatter_text.split('\n'):
            if ':' in line:
                key, value = line.split(':', 1)
                metadata[key.strip()] = value.strip()

        # Extract description from frontmatter or content
        description = metadata.get('description', '')
        if not description:
            # Try to get first heading or paragraph
            body = content[end_idx + 3:].strip()
            lines = [l for l in body.split('\n') if l.strip() and not l.strip().startswith('#')]
            if lines:
                description = lines[0].strip()[:100]
            else:
                description = f"Skill: {skill_file.stem}"

        return {
            "description": description,
            "file_path": str(skill_file),
            "name": metadata.get('name', skill_file.stem)
        }
    except Exception as e:
        print(f"Warning: Failed to parse skill file {skill_file}: {e}")
        return None

class SubAgent:
    """Sub-agent with limited scope and iteration count."""

    MAX_ITERATIONS = 30

    def __init__(self):
        self.client = AsyncAnthropic(
            api_key=ANTHROPIC_API_KEY,
            base_url=ANTHROPIC_BASE_URL
        )
        self.messages: list[dict[str, Any]] = []
        self.iteration_count = 0

    async def read_file(self, file_path: str) -> str:
        """Read a file and return its contents."""
        try:
            with open(file_path, 'r', encoding='utf-8') as f:
                content = f.read()
            return content if content else "(empty file)"
        except FileNotFoundError:
            return f"Error: File not found: {file_path}"
        except PermissionError:
            return f"Error: Permission denied: {file_path}"
        except Exception as e:
            return f"Error: {str(e)}"

    async def write_file(self, file_path: str, content: str) -> str:
        """Write content to a file."""
        try:
            parent_dir = os.path.dirname(file_path)
            if parent_dir and not os.path.exists(parent_dir):
                os.makedirs(parent_dir)
            with open(file_path, 'w', encoding='utf-8') as f:
                f.write(content)
            return f"Successfully wrote to {file_path}"
        except PermissionError:
            return f"Error: Permission denied: {file_path}"
        except Exception as e:
            return f"Error: {str(e)}"

    async def edit_file(self, file_path: str, old_string: str, new_string: str) -> str:
        """Edit a file by replacing old_string with new_string."""
        try:
            with open(file_path, 'r', encoding='utf-8') as f:
                content = f.read()
            if old_string not in content:
                return f"Error: String not found in file: {old_string[:50]}..."
            new_content = content.replace(old_string, new_string, 1)
            with open(file_path, 'w', encoding='utf-8') as f:
                f.write(new_content)
            return f"Successfully edited {file_path}"
        except FileNotFoundError:
            return f"Error: File not found: {file_path}"
        except PermissionError:
            return f"Error: Permission denied: {file_path}"
        except Exception as e:
            return f"Error: {str(e)}"

    async def execute_bash(self, command: str) -> str:
        """Execute a bash command and return the output."""
        try:
            result = subprocess.run(
                command,
                shell=True,
                capture_output=True,
                text=True,
                timeout=300
            )
            output = result.stdout
            if result.stderr:
                output += f"\nSTDERR: {result.stderr}"
            if result.returncode != 0:
                output += f"\nExit Code: {result.returncode}"
            return output.strip() if output.strip() else "(no output)"
        except subprocess.TimeoutExpired:
            return "Error: Command timed out after 300 seconds"
        except Exception as e:
            return f"Error: {str(e)}"

    async def call_tool(self, tool_name: str, tool_input: dict) -> str:
        """Handle tool calls for sub-agent."""
        if tool_name == "read_file":
            return await self.read_file(tool_input.get("file_path", ""))
        elif tool_name == "write_file":
            return await self.write_file(
                tool_input.get("file_path", ""),
                tool_input.get("content", "")
            )
        elif tool_name == "edit_file":
            return await self.edit_file(
                tool_input.get("file_path", ""),
                tool_input.get("old_string", ""),
                tool_input.get("new_string", "")
            )
        elif tool_name == "bash":
            return await self.execute_bash(tool_input.get("command", ""))
        else:
            return f"Error: Unknown tool '{tool_name}'"

    async def run(self, prompt: str) -> str:
        """
        Run the sub-agent with limited iterations.
        Returns the final result or error if max iterations reached.
        """
        self.messages = [{"role": "user", "content": prompt}]
        self.iteration_count = 0

        while self.iteration_count < self.MAX_ITERATIONS:
            self.iteration_count += 1

            response = await self.client.messages.create(
                model=MODEL_ID,
                max_tokens=4096,
                system=SUBAGENT_SYSTEM_PROMPT,
                tools=SUBAGENT_TOOLS,
                messages=self.messages
            )

            stop_reason = response.stop_reason
            self.messages.append({
                "role": "assistant",
                "content": response.content
            })

            if stop_reason == "tool_use":
                tool_results = []
                for block in response.content:
                    if block.type == "tool_use":
                        tool_name = block.name
                        tool_input = block.input
                        tool_id = block.id

                        result = await self.call_tool(tool_name, tool_input)

                        tool_results.append({
                            "type": "tool_result",
                            "tool_use_id": tool_id,
                            "content": result
                        })

                self.messages.append({
                    "role": "user",
                    "content": tool_results
                })
                continue

            # No more tool calls - extract final answer
            final_answer = ""
            for block in response.content:
                if hasattr(block, 'text'):
                    final_answer += block.text

            return final_answer.strip() if final_answer.strip() else "Task completed."

        # Max iterations reached
        return f"Error: Sub-agent reached maximum iterations ({self.MAX_ITERATIONS}) without completing the task."

    def _count_tool_results(self) -> int:
        """
        Count the number of complete tool interaction rounds.
        A complete round is: assistant message with tool_use + user message with tool_result.

        Returns:
            int: Number of complete tool interaction rounds
        """
        count = 0
        for i, msg in enumerate(self.messages):
            if msg["role"] == "assistant" and isinstance(msg.get("content"), list):
                # Check if this assistant message contains tool_use blocks
                has_tool_use = False
                for block in msg["content"]:
                    if hasattr(block, 'type') and block.type == "tool_use":
                        has_tool_use = True
                        break
                    elif isinstance(block, dict) and block.get("type") == "tool_use":
                        has_tool_use = True
                        break
                if has_tool_use and i + 1 < len(self.messages):
                    next_msg = self.messages[i + 1]
                    if next_msg["role"] == "user" and isinstance(next_msg.get("content"), list):
                        # Check if next message contains tool_result
                        has_tool_result = False
                        for block in next_msg["content"]:
                            if isinstance(block, dict) and block.get("type") == "tool_result":
                                has_tool_result = True
                                break
                        if has_tool_result:
                            count += 1
        return count

    def _extract_tool_names_from_history(self, messages_to_compress: list) -> list[str]:
        """
        Extract tool names from assistant messages that contain tool_use blocks.

        Args:
            messages_to_compress: List of messages to analyze

        Returns:
            list[str]: List of tool names used in those messages
        """
        tool_names = []
        for msg in messages_to_compress:
            if msg["role"] == "assistant" and isinstance(msg["content"], list):
                for block in msg["content"]:
                    if hasattr(block, 'type') and block.type == "tool_use":
                        tool_names.append(block.name)
                    elif isinstance(block, dict) and block.get("type") == "tool_use":
                        tool_names.append(block.get("name", "unknown"))
        return tool_names

    async def compress_context(self, summary: str) -> str:
        """
        Compress historical context by replacing older tool interactions with a summary node.

        This method:
        1. Identifies complete tool_use + tool_result pairs in messages
        2. Preserves the most recent CONTEXT_KEEP_RECENT_ROUNDS complete tool interactions
        3. Replaces older interactions with a compressed memory node containing the summary

        Args:
            summary: A concise summary of what was accomplished and key findings

        Returns:
            str: Status message about compression result
        """
        if len(self.messages) < 4:  # Need at least: user + assistant(tool_use) + user(tool_result) + assistant
            return "❌ Context too short to compress (need at least 4 messages)"

        # Count complete tool interaction rounds (assistant with tool_use + user with tool_result)
        tool_round_indices = []  # List of (assistant_idx, tool_result_idx) pairs
        for i, msg in enumerate(self.messages):
            if msg["role"] == "assistant" and isinstance(msg.get("content"), list):
                # Check if this assistant message contains tool_use blocks
                has_tool_use = False
                for block in msg["content"]:
                    if hasattr(block, 'type') and block.type == "tool_use":
                        has_tool_use = True
                        break
                if has_tool_use and i + 1 < len(self.messages):
                    next_msg = self.messages[i + 1]
                    if next_msg["role"] == "user" and isinstance(next_msg.get("content"), list):
                        # Check if next message contains tool_result
                        has_tool_result = False
                        for block in next_msg["content"]:
                            if isinstance(block, dict) and block.get("type") == "tool_result":
                                has_tool_result = True
                                break
                        if has_tool_result:
                            tool_round_indices.append((i, i + 1))

        if len(tool_round_indices) <= CONTEXT_KEEP_RECENT_ROUNDS:
            return f"❌ Only {len(tool_round_indices)} complete tool rounds, nothing to compress (minimum {CONTEXT_KEEP_RECENT_ROUNDS + 1} needed)"

        # Keep the last CONTEXT_KEEP_RECENT_ROUNDS complete rounds
        # Find the cutoff point: the assistant message index of the first round to keep
        first_kept_assistant_idx = tool_round_indices[-CONTEXT_KEEP_RECENT_ROUNDS][0]

        # Messages to keep (from first_kept_assistant_idx onwards)
        messages_to_keep = self.messages[first_kept_assistant_idx:]

        # Messages to compress (everything before first_kept_assistant_idx)
        messages_to_compress = self.messages[:first_kept_assistant_idx]

        # Extract tool names from compressed history
        tool_names = self._extract_tool_names_from_history(messages_to_compress)

        # Build compressed summary message
        compressed_tools_summary = ""
        if tool_names:
            tool_counts = {}
            for name in tool_names:
                tool_counts[name] = tool_counts.get(name, 0) + 1
            compressed_tools_summary = "\n压缩的工具调用记录:\n"
            for name, count in sorted(tool_counts.items(), key=lambda x: -x[1]):
                compressed_tools_summary += f"  - {name}: 执行 {count} 次\n"

        compressed_message = {
            "role": "user",
            "content": f"""[系统已压缩上下文摘要]

以下是之前对话历史的压缩摘要：

{summary}

{compressed_tools_summary}

原始的详细工具调用结果已被压缩。如果需要特定历史细节，请根据摘要内容推断或重新执行相关查询。
"""
        }

        # Build new messages list
        # Structure: [original_user_request] + [compressed_summary] + [kept_messages]
        new_messages = []

        # Keep the first user message (original request)
        if self.messages and self.messages[0]["role"] == "user":
            first_user_content = self.messages[0]["content"]
            # Only keep if it's not a tool_result message
            if isinstance(first_user_content, str) or (
                isinstance(first_user_content, list) and
                not any(isinstance(b, dict) and b.get("type") == "tool_result" for b in first_user_content)
            ):
                new_messages.append(self.messages[0])

        new_messages.append(compressed_message)
        new_messages.extend(messages_to_keep)

        # Calculate compression stats
        old_count = len(self.messages)
        new_count = len(new_messages)
        reduced = old_count - new_count
        tool_rounds_compressed = len(tool_round_indices) - CONTEXT_KEEP_RECENT_ROUNDS

        # Update messages
        self.messages = new_messages

        print(f"\n🗜️ Context Compressed:")
        print(f"   📊 Before: {old_count} messages, {len(tool_round_indices)} tool rounds")
        print(f"   📊 After: {new_count} messages")
        print(f"   📊 Reduced: {reduced} messages compressed ({tool_rounds_compressed} tool rounds)")
        print(f"   🔒 Preserved: {CONTEXT_KEEP_RECENT_ROUNDS} recent tool rounds")

        return f"✅ Context compressed successfully. {reduced} messages reduced to 1 summary node. Preserved {CONTEXT_KEEP_RECENT_ROUNDS} recent tool interaction rounds."

    async def auto_compress_context(self) -> tuple[bool, str]:
        """
        Automatically compress context when tool_result count exceeds threshold.

        This is called from run_loop when the threshold is exceeded.
        It generates an automatic summary based on the todo list and compresses older context.

        Returns:
            tuple[bool, str]: (compression_occurred, status_message)
        """
        tool_result_count = self._count_tool_results()

        if tool_result_count <= CONTEXT_COMPRESSION_THRESHOLD:
            return (False, "No compression needed")

        # Generate automatic summary from todo list and recent activity
        todo_summary = ""
        if TODO_LIST:
            completed = [t['description'] for t in TODO_LIST if t['status'] == '完成']
            pending = [t['description'] for t in TODO_LIST if t['status'] == '等待']
            failed = [t['description'] for t in TODO_LIST if t['status'] == '失败']

            todo_summary = "当前任务进度:\n"
            if completed:
                todo_summary += f"已完成: {', '.join(completed[:5])}\n"
            if pending:
                todo_summary += f"待处理: {', '.join(pending[:5])}\n"
            if failed:
                todo_summary += f"失败: {', '.join(failed[:3])}\n"

        # Extract tool names from entire history for auto-summary
        tool_names = self._extract_tool_names_from_history(self.messages)
        tool_summary_parts = []
        if tool_names:
            tool_counts = {}
            for name in tool_names:
                tool_counts[name] = tool_counts.get(name, 0) + 1
            tool_summary_parts.append("曾执行工具调用:")
            for name, count in sorted(tool_counts.items(), key=lambda x: -x[1])[:10]:
                tool_summary_parts.append(f"  {name}: {count}次")

        auto_summary = f"""[自动压缩摘要 - {tool_result_count}轮工具交互]

{todo_summary}
{chr(10).join(tool_summary_parts) if tool_summary_parts else ''}

系统检测到上下文工具交互数量超过阈值({CONTEXT_COMPRESSION_THRESHOLD})，自动触发压缩。
"""

        # Perform compression
        result = await self.compress_context(auto_summary)
        return (True, result)

    async def run_loop(self, user_input: str) -> str:
        """
        Main async control loop following ReAct pattern.
        - Request LLM with user input
        - Check stop_reason for tool_use
        - Execute tools and append results to messages
        - Continue until no more tool calls
        - Monitor: Inject reminder if 3 consecutive rounds without update_todo
        """
        # Initialize messages with user input
        self.messages = [
            {"role": "user", "content": user_input}
        ]

        # Reset the consecutive no-update counter at the start of each session
        self.consecutive_no_todo_updates = 0

        # Get dynamic system prompt with current skills
        system_prompt = get_system_prompt()

        while True:
            # Request LLM
            response = await self.client.messages.create(
                model=MODEL_ID,
                max_tokens=4096,
                system=system_prompt,
                tools=TOOLS,
                messages=self.messages
            )

            # Check stop_reason
            stop_reason = response.stop_reason

            # Append assistant response to messages
            self.messages.append({
                "role": "assistant",
                "content": response.content
            })

            # If tool_use, process and continue loop
            if stop_reason == "tool_use":
                # Find and process all tool_use blocks
                tool_results = []

                # Track if update_todo was called in this round
                todo_updated_this_round = False

                for block in response.content:
                    if block.type == "tool_use":
                        tool_name = block.name
                        tool_input = block.input
                        tool_id = block.id

                        # Check if update_todo is being called
                        if tool_name == "update_todo":
                            todo_updated_this_round = True

                        # Execute the tool
                        result = await self.process_tool_use(tool_name, tool_input)

                        # Build tool result following Anthropic tool-use spec
                        tool_results.append({
                            "type": "tool_result",
                            "tool_use_id": tool_id,
                            "content": result
                        })

                # Update counter based on whether update_todo was called
                if todo_updated_this_round:
                    self.consecutive_no_todo_updates = 0
                else:
                    self.consecutive_no_todo_updates += 1

                # Append tool results as user message
                self.messages.append({
                    "role": "user",
                    "content": tool_results
                })

                # ========== Context Compression Check ==========
                # Check if automatic compression should be triggered
                tool_result_count = self._count_tool_results()
                if tool_result_count > CONTEXT_COMPRESSION_THRESHOLD:
                    print(f"\n⚠️ Context threshold reached: {tool_result_count} tool_results > {CONTEXT_COMPRESSION_THRESHOLD}")
                    print("🗜️ Triggering automatic context compression...")

                    compressed, compress_result = await self.auto_compress_context()
                    if compressed:
                        print(f"✅ {compress_result}")
                    else:
                        print(f"ℹ️ {compress_result}")

                # ========== Todo Reminder Check ==========
                # Check if we need to inject a reminder
                if self.consecutive_no_todo_updates >= self.TODO_REMINDER_THRESHOLD:
                    print(f"\n⚠️ WARNING: {self.consecutive_no_todo_updates} consecutive rounds without update_todo call")
                    print("📋 Injecting reminder to sync task progress...")

                    # Inject reminder as a separate user message (not as tool_result)
                    reminder_msg = (
                        "【系统提醒】您已连续多轮推理未更新任务清单。\n"
                        "为了保持推理一致性，请立即使用 update_todo 工具同步当前任务进度。\n"
                        "当前状态：\n"
                        f"- 待处理任务数: {len([t for t in TODO_LIST if t['status'] == '等待'])}\n"
                        f"- 已完成任务数: {len([t for t in TODO_LIST if t['status'] == '完成'])}\n"
                        f"- 失败任务数: {len([t for t in TODO_LIST if t['status'] == '失败'])}\n"
                        "请审视当前进度并更新任务状态。"
                    )
                    self.messages.append({
                        "role": "user",
                        "content": reminder_msg
                    })

                    # Reset counter after reminder
                    self.consecutive_no_todo_updates = 0

                # Continue the loop for next iteration
                continue

            # No more tool calls - extract and return final answer
            final_answer = ""
            for block in response.content:
                if hasattr(block, 'text'):
                    final_answer += block.text

            return final_answer.strip() if final_answer.strip() else "Task completed."

    async def run_interactive(self):
        """Run the interactive CLI session."""
        print("=" * 50)
        print("ReAct CLI Agent (AsyncAnthropic) with Skills Support")
        print(f"Model: {MODEL_ID}")
        print(f"Skills Directory: {SKILLS_DIR}")
        print(f"Available Skills: {list(self.available_skills.keys())}")
        print("Type 'exit' or 'quit' to stop")
        print("Type 'skills' to list available skills")
        print("=" * 50)

        while True:
            try:
                # Get user input
                user_input = input("\n👤 You: ").strip()

                if not user_input:
                    continue

                if user_input.lower() in ("exit", "quit"):
                    print("\n👋 Goodbye!")
                    break

                if user_input.lower() == "skills":
                    print("\n📚 Available Skills:")
                    for name, info in self.available_skills.items():
                        print(f"   - {name}: {info['description']}")
                    continue

                # Run the agent loop
                print("\n🤖 Agent thinking...")
                result = await self.run_loop(user_input)

                # Print final answer
                print(f"\n✅ Final Answer:\n{result}")

            except KeyboardInterrupt:
                print("\n\n👋 Goodbye!")
                break
            except Exception as e:
                print(f"\n❌ Error: {str(e)}")
